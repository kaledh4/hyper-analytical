import yfinance as yf
import pandas as pd
import pandas_datareader.data as web # For FRED Data
import numpy as np
import requests
import os
import json
from datetime import datetime, timedelta

# --- CONFIG ---
# Get keys from GitHub Secrets
OPENROUTER_KEY = os.environ.get("OPENROUTER_API_KEY")
TELEGRAM_TOKEN = os.environ.get("TELEGRAM_BOT_TOKEN")
TELEGRAM_CHAT_ID = os.environ.get("TELEGRAM_CHAT_ID")
SITE_URL = "https://<YOUR_USERNAME>.github.io/<REPO_NAME>/"

# --- DATA INGESTION (THE "MACRO" VIEW) ---
def get_macro_data():
    """Fetches Fed Funds, CPI, and Yields from St. Louis Fed (FRED) via Pandas Datareader."""
    start = datetime.now() - timedelta(days=365*5)
    
    try:
        # DGS10 = 10-Year Treasury, DGS2 = 2-Year Treasury, FEDFUNDS = Interest Rate, CPIAUCSL = CPI
        macro = web.DataReader(['DGS10', 'DGS2', 'FEDFUNDS', 'CPIAUCSL'], 'fred', start, datetime.now())
        
        # Calculate Yield Curve (10Y - 2Y) - The "Recession Indicator"
        macro['Yield_Curve'] = macro['DGS10'] - macro['DGS2']
        return macro
    except Exception as e:
        print(f"FRED Data Error (Using fallbacks): {e}")
        return pd.DataFrame()

def get_crypto_data(ticker):
    """Fetches Crypto Data and calculates Heikin-Ashi and BMSB."""
    df = yf.download(ticker, period="5y", interval="1wk")
    if isinstance(df.columns, pd.MultiIndex): df = df.xs(ticker, axis=1, level=1)
    
    # 1. Bull Market Support Band (Weekly)
    df['SMA_20'] = df['Close'].rolling(window=20).mean()
    df['EMA_21'] = df['Close'].ewm(span=21, adjust=False).mean()
    
    # 2. Heikin-Ashi (Trend Smoothing)
    df['HA_Close'] = (df['Open'] + df['High'] + df['Low'] + df['Close']) / 4
    # (Simplified HA Open for speed)
    df['HA_Open'] = (df['Open'].shift(1) + df['Close'].shift(1)) / 2
    df['Trend'] = np.where(df['HA_Close'] > df['HA_Open'], "Bullish", "Bearish")
    
    return df

# --- THE "RISK METRIC" CALCULATOR ---
def calculate_risk_metric(df):
    """
    Approximation of Professional Risk Metric:
    Uses Logarithmic Regression deviation normalized (0-1) over a 4-year cycle.
    """
    # Prepare Log Log Data
    data = df.copy().dropna()
    data['log_price'] = np.log(data['Close'])
    data['time_idx'] = np.arange(1, len(data) + 1)
    data['log_time'] = np.log(data['time_idx'])
    
    # Fit Regression
    slope, intercept = np.polyfit(data['log_time'], data['log_price'], 1)
    data['fair_value'] = np.exp(intercept + slope * data['log_time'])
    
    # Calculate % Deviation from "Fair Value"
    data['deviation'] = (data['Close'] - data['fair_value']) / data['fair_value']
    
    # Normalize (0 to 1) based on Rolling 200-Week Window (approx 4 years)
    roll_min = data['deviation'].rolling(window=200).min()
    roll_max = data['deviation'].rolling(window=200).max()
    
    data['risk'] = (data['deviation'] - roll_min) / (roll_max - roll_min)
    
    # Clip to 0-1 just in case
    data['risk'] = data['risk'].clip(0, 1)
    return data['risk'].iloc[-1], data['risk'].iloc[-2] # Current and Prev

# --- THE "PROFESSIONAL" ANALYST (LLM) ---
def generate_cowen_analysis(btc, eth, dxy, macro_data, risk):
    
    # Construct "The Narrative"
    yield_curve = macro_data['Yield_Curve'].iloc[-1] if not macro_data.empty else -0.5
    fed_rate = macro_data['FEDFUNDS'].iloc[-1] if not macro_data.empty else 5.25
    btc_price = btc['Close'].iloc[-1]
    sma_20 = btc['SMA_20'].iloc[-1]
    ema_21 = btc['EMA_21'].iloc[-1]
    
    eth_btc_ratio = eth['Close'].iloc[-1] / btc['Close'].iloc[-1]
    
    prompt = f"""
    You are a professional crypto market analyst at Hyper Analytical. Your goal is to provide a DRY, DATA-DRIVEN, and REDUNDANT daily update.
    
    CORE RULES:
    1.  **No Hype:** Never use words like "moon," "surge," or "skyrocket." Use "expansion," "valuation," and "move back to the range."
    2.  **Redundancy:** Explain the point, then explain it again using a different metric (e.g., "If we look at price... now if we look at the logarithmic regression...").
    3.  **Catchphrases:** Use "Dubious speculation," "Don't schedule your Forbes interview," "What is the primary driver?", "Home to the Bull Market Support Band."
    4.  **Macro Focus:** You MUST discuss the Fed and the DXY (Dollar Index). The crypto market does not exist in a vacuum.
    
    DATA FOR TODAY:
    -   **Bitcoin Price:** ${btc_price:.2f}
    -   **Bull Market Support Band:** ${sma_20:.2f} (20W SMA) and ${ema_21:.2f} (21W EMA).
        * (Context: If Price < Band, we are in the accumulation phase/bear market. If > Band, bull market).
    -   **Risk Metric (0.0-1.0):** {risk:.2f}. (0.0 = Depression, 1.0 = Euphoria).
    -   **ETH/BTC Ratio:** {eth_btc_ratio:.5f}. (Context: If trending down, say "ALTS are bleeding against Bitcoin").
    -   **DXY (Dollar):** {dxy['Close'].iloc[-1]:.2f}.
    -   **Yield Curve (10Y-2Y):** {yield_curve:.2f} (Negative = Inverted/Recession warning).
    -   **Fed Funds Rate:** {fed_rate:.2f}%.
    
    STRUCTURE OF THE UPDATE:
    1.  **The Bitcoin Valuation:** Discuss the price relative to the BMSB. Are we testing it? Are we far above it?
    2.  **The Risk Analysis:** Discuss the Risk Metric. Is it wise to deploy capital now? (Cash is a position).
    3.  **The Macro Headwinds:** Discuss DXY and the Yield Curve. Mention how the Fed's higher-for-longer policy impacts risk assets.
    4.  **Altcoin Reckoning:** Look at ETH/BTC. Remind users that in a pre-halving or QT year, ALTS bleed to BTC.
    5.  **Conclusion:** Wrap up with "Hyper Analytical."
    """
    
    payload = {
        "model": "google/gemini-2.0-flash-lite-preview-02-05:free", # Free High Context Model
        "messages": [{"role": "user", "content": prompt}],
        "temperature": 0.3 # Low temp for dry/analytical tone
    }
    
    try:
        res = requests.post(
            "https://openrouter.ai/api/v1/chat/completions",
            headers={"Authorization": f"Bearer {OPENROUTER_KEY}"},
            json=payload
        )
        return res.json()['choices'][0]['message']['content']
    except:
        return "Analysis unavailable due to API constraints. But the math remains the same."

# --- MAIN EXECUTION ---
def main():
    print("Fetching Macro Data (FRED)...")
    macro = get_macro_data()
    
    print("Fetching Crypto Data...")
    btc = get_crypto_data("BTC-USD")
    eth = get_crypto_data("ETH-USD")
    dxy = get_crypto_data("DX-Y.NYB") # Dollar Index
    
    # Calculate Risk
    risk_curr, risk_prev = calculate_risk_metric(btc)
    
    print("Consulting the Analyst...")
    analysis = generate_cowen_analysis(btc, eth, dxy, macro, risk_curr)
    
    # Prepare JSON for Frontend
    dashboard_data = {
        "date": datetime.now().strftime("%B %d, %Y"),
        "btc_price": float(btc['Close'].iloc[-1]),
        "bmsb": {
            "sma_20": float(btc['SMA_20'].iloc[-1]),
            "ema_21": float(btc['EMA_21'].iloc[-1])
        },
        "risk_metric": {
            "current": float(risk_curr),
            "history": btc['risk'].tail(52).tolist() # Send last year of risk data for charting
        },
        "macro": {
            "dxy": float(dxy['Close'].iloc[-1]),
            "yield_inversion": float(macro['Yield_Curve'].iloc[-1]) if not macro.empty else -0.4,
            "fed_rate": float(macro['FEDFUNDS'].iloc[-1]) if not macro.empty else 5.25
        },
        "commentary": analysis
    }
    
    # Write Data
    with open("dashboard_data.json", "w") as f:
        json.dump(dashboard_data, f)
        
    # Telegram Alert
    msg = f"**Into The Cryptoverse (Daily Update)**\n\n" \
          f"Bitcoin: ${dashboard_data['btc_price']:,.0f}\n" \
          f"Risk Metric: {dashboard_data['risk_metric']['current']:.2f}\n" \
          f"BMSB: ${dashboard_data['bmsb']['sma_20']:,.0f} - ${dashboard_data['bmsb']['ema_21']:,.0f}\n" \
          f"Yield Curve: {dashboard_data['macro']['yield_inversion']}%\n\n" \
          f"Read the full analysis: {SITE_URL}"
          
    requests.post(f"https://api.telegram.org/bot{TELEGRAM_TOKEN}/sendMessage", 
                  data={"chat_id": TELEGRAM_CHAT_ID, "text": msg, "parse_mode": "Markdown"})

if __name__ == "__main__":
    main()
